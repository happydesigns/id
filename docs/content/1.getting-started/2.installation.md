---

Install the package in a Nuxt project.

::code-group
```bash [pnpm]
pnpm add @happydesigns/id
```

```bash [npm]
npm install @happydesigns/id
```

```bash [yarn]
yarn add @happydesigns/id
```

```bash [bun]
bun add @happydesigns/id
```
::

## Extend the layer

Use the layer when the project wants the default identity runtime.

```ts [nuxt.config.ts]
export default defineNuxtConfig({
  extends: ['@happydesigns/id/nuxt']
})
```

## Use the module

Use the module when the project wants explicit module options.

```ts [nuxt.config.ts]
export default defineNuxtConfig({
  modules: ['@happydesigns/id/module'],
  id: {
    defaultTheme: 'client',
    themes: []
  }
})
```

## Configure a theme

Add theme packs in `app.config.ts`.

```ts [app.config.ts]
import { defineBrandTheme } from '@happydesigns/id'

const clientTheme = defineBrandTheme({
  name: 'client',
  label: 'Client',
  semanticColors: {
    primary: 'blue',
    neutral: 'slate'
  },
  cssVariables: {
    light: {
      '--ui-bg': 'white',
      '--ui-text': 'var(--ui-color-slate-700)'
    }
  }
})

export default defineAppConfig({
  id: {
    defaultTheme: clientTheme.name,
    themes: [clientTheme]
  },
  ui: {
    colors: clientTheme.semanticColors
  }
})
```
